/*
    Craft recipes. Display names come from Translate/<LANG>/Recipes.json, keyed by recipe name.
    Callbacks are in lua/shared/RenewableEnergy/RecipeCode.lua.

    The recipes live in module Base, prefixed against name clashes: the game looks a recipe up
    by its bare name when researching an item and in a magazine's recipe list, and a bare name
    resolves only in Base (ScriptBucketCollection.getScript).

    Conventions follow vanilla's electrical recipes: tools mode:keep with
    flags[Prop1;MayDegradeLight] and a table within reach (AnySurfaceCraft). Base.BlowTorch
    inputs count uses on purpose.

    Learning: NeedToBeLearn recipes come from the Homestead Power magazines, from AutoLearnAny,
    or from researching a found item (the item's output recipe, plus its ResearchableRecipes).
    Taking a panel mount apart needs no learning.
*/
module Base
{
    /**************************** Stage 0 ***********************************/

    /* The stage-0 bank. Its inverter goes in the box's slot, so the recipe takes none. */
    craftRecipe RenewableEnergy_MakeBatteryBox
    {
        timedAction = MakingElectrical,
        time = 200,
        Tags = AnySurfaceCraft,
        category = Electrical,
        NeedToBeLearn = true,
        AutoLearnAny = Electricity:2,
        SkillRequired = Electricity:1;Woodwork:1,
        xpAward = Electricity:10;Woodwork:5,
        inputs
        {
            item 1 tags[base:hammer] mode:keep flags[Prop1;MayDegradeLight],
            item 1 tags[base:screwdriver] mode:keep flags[MayDegradeLight],
            item 4 [Base.Plank],
            item 8 [Base.Nails],
            item 2 [Base.ElectricWire],
        }
        outputs
        {
            item 1 RenewableEnergy.BatteryBox,
        }
    }
