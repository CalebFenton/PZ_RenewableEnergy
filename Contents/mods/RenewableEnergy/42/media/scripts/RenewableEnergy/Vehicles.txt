module Base
{
    /* In module Base, as vanilla's vehicles are: templates and model files are looked up by bare
       name there. art/models/models.py writes the mesh and its textures; the body is built in the
       frame of vanilla's Trailer, so the numbers below are that trailer's, multiplied out by its
       model scale, and read in metres.

       vehicle_multiuv_noreflect rather than its norandom twin: the damage shells take their hue
       from the paint, and their saturation and value as offsets from it, outside the Head, Tail,
       window and lamp zones (vehicle_common.frag addDamage). Norandom's paint is the skin texel,
       so on the orange paint every dent stays orange; this one's is the vehicle's colour, whose
       forcedColor below cancels the offsets, so the shells draw their own grey. The skin is
       opaque, so that colour never repaints it. */
    model RenewableEnergyArrowBoardTrailer
    {
        mesh = vehicles/RenewableEnergy_ArrowBoardTrailer,
        shader = vehicle_multiuv_noreflect,
    }

    /* Part models cut from the same bake as the body, in the body's frame, so they sit in place at
       a part offset of zero and draw with the body's skin, mask, rust, damage and blood. The lid
       keeps the Boot zone that shows its damage. It is skinned to a bone on its hinge, with the
       clip that swings it (art/models/models.py), and only a model that isn't static gets an
       animation player to play that clip (BaseVehicle.ModelInfo). */
    model RenewableEnergyArrowBoardTrailerLid
    {
        mesh = vehicles/RenewableEnergy_ArrowBoardTrailer_Lid,
        shader = vehicle_multiuv_noreflect,
        static = false,
    }
    /* The padlock draws only on a shut lid, so it needs no bone and stays static. */
    model RenewableEnergyArrowBoardTrailerPadlock
    {
        mesh = vehicles/RenewableEnergy_ArrowBoardTrailer_Padlock,
        shader = vehicle_multiuv_noreflect,
    }
    model RenewableEnergyArrowBoardTrailerPanelLeft
    {
        mesh = vehicles/RenewableEnergy_ArrowBoardTrailer_PanelLeft,
        shader = vehicle_multiuv_noreflect,
    }
    model RenewableEnergyArrowBoardTrailerPanelLeftCracked
    {
        mesh = vehicles/RenewableEnergy_ArrowBoardTrailer_PanelLeftCracked,
        shader = vehicle_multiuv_noreflect,
    }
    model RenewableEnergyArrowBoardTrailerPanelLeftLightCracked
    {
        mesh = vehicles/RenewableEnergy_ArrowBoardTrailer_PanelLeftLightCracked,
        shader = vehicle_multiuv_noreflect,
    }
    model RenewableEnergyArrowBoardTrailerPanelRight
    {
        mesh = vehicles/RenewableEnergy_ArrowBoardTrailer_PanelRight,
        shader = vehicle_multiuv_noreflect,
    }
    model RenewableEnergyArrowBoardTrailerPanelRightCracked
    {
        mesh = vehicles/RenewableEnergy_ArrowBoardTrailer_PanelRightCracked,
        shader = vehicle_multiuv_noreflect,
    }
    model RenewableEnergyArrowBoardTrailerPanelRightLightCracked
    {
        mesh = vehicles/RenewableEnergy_ArrowBoardTrailer_PanelRightLightCracked,
        shader = vehicle_multiuv_noreflect,
    }

    vehicle RenewableEnergyArrowBoardTrailer
    {
        neverSpawnKey = true,
        mechanicType = 1,
        offRoadEfficiency = 0.8,
        engineRepairLevel = 4,
        playerDamageProtection = 0.8,
        model
        {
            file = RenewableEnergyArrowBoardTrailer,
            scale = 1.0,
            offset = 0.0 0.34 0.25,
        }
        skin
        {
            texture = Vehicles/RenewableEnergy_ArrowBoardTrailer,
        }
        /* Hue, saturation, value. Half saturation and value add nothing to the damage shells', so
           they keep their grey; ArrowBoard.initFrame sets it again on trailers spawned before. */
        forcedColor = 0.0 0.5 0.5,
        textureMask = Vehicles/RenewableEnergy_ArrowBoardTrailer_Mask,
        textureLights = Vehicles/RenewableEnergy_ArrowBoardTrailer_Lights,
        /* The multi-UV shader draws blood from the overlays: where it grows first, then its colour. */
        textureDamage1Overlay = Vehicles/RenewableEnergy_ArrowBoardTrailer_BloodMask,
        textureDamage2Overlay = Vehicles/RenewableEnergy_ArrowBoardTrailer_Blood,
        textureDamage1Shell = Vehicles/RenewableEnergy_ArrowBoardTrailer_Damage1,
        textureDamage2Shell = Vehicles/RenewableEnergy_ArrowBoardTrailer_Damage2,
        textureRust = Vehicles/RenewableEnergy_ArrowBoardTrailer_Rust,
        extents = 1.14 0.4 1.7,
        mass = 300,
        physicsChassisShape = 1.14 0.4 1.7,
        centerOfMassOffset = 0.0 0.4 -0.33,
        shadowExtents = 1.5 2.4,
        shadowOffset = 0.0 -0.1,
        engineForce = 3600,
        engineQuality = 60,
        engineLoudness = 55,
        maxSpeed = 70.0,
        brakingForce = 1,
        stoppingMovementForce = 2.0,
        rollInfluence = 1.0,
        steeringIncrement = 0.03,
        steeringClamp = 0.3,
        suspensionStiffness = 40.0,
        suspensionCompression = 2.83,
        suspensionDamping = 2.88,
        maxSuspensionTravelCm = 18.2,
        suspensionRestLength = 0.364,
        wheelFriction = 4.0,
        frontEndHealth = 150,
        rearEndHealth = 150,
        seats = 2,
        /* The tongue's two rails and coupler, then the deck. */
        physics box
        {
            offset = 0.2 0.18 0.98,
            extents = 0.1 0.1 0.98,
            rotate = 0.0 -20.0 0.0,
        }
        physics box
        {
            offset = -0.2 0.18 0.98,
            extents = 0.1 0.1 0.98,
            rotate = 0.0 20.0 0.0,
        }
        physics sphere
        {
            offset = 0.0 0.18 1.38,
            radius = 0.1,
        }
        physics box
        {
            offset = 0.0 0.39 -0.34,
            extents = 1.16 0.4 1.74,
            rotate = 0.0 0.0 0.0,
        }
        useChassisPhysicsCollision = false,
        wheel FrontLeft
        {
            front = true,
            offset = 0.68 -0.34 -0.55,
            radius = 0.273,
            width = 0.2,
        }
        wheel FrontRight
        {
            front = true,
            offset = -0.68 -0.34 -0.55,
            radius = 0.273,
            width = 0.2,
        }
        /* Where a character stands to reach the battery box on the tongue: either side of it or
           in front. */
        area RenewableEnergyBatteryBox
        {
            xywh = 0.0 0.85 2.0 1.0,
        }
        area TireFrontLeft
        {
            xywh = 0.97 -0.31 0.8 0.86,
        }
        area TireFrontRight
        {
            xywh = -0.97 -0.31 0.8 0.86,
        }
        /* Where a character stands to take a panel off or put one on: beside its half of the frame. */
        area RenewableEnergyPanelLeft
        {
            xywh = 0.97 0.45 0.8 1.0,
        }
        area RenewableEnergyPanelRight
        {
            xywh = -0.97 0.45 0.8 1.0,
        }
        /* Where a character stands to wash each side's blood off, as ISWashVehicle finds them by
           these names (Front, Rear, Left, Right), and to weld the front and the back of the frame:
           around the coupler, so the sides of the tongue stay reachable with a car on the hitch;
           behind the deck; beside each side. No passenger names the seat areas, so they seat no
           one. */
        area Engine
        {
            xywh = 0.0 1.6 2.0 0.8,
        }
        area TruckBed
        {
            xywh = 0.0 -1.55 2.0 0.7,
        }
        area SeatLeft
        {
            xywh = 0.97 -0.3 0.8 1.8,
        }
        area SeatRight
        {
            xywh = -0.97 -0.3 0.8 1.8,
        }
        template = Tire/part/TireFrontLeft,
        template = Tire/part/TireFrontRight,
        template = Suspension/part/SuspensionFrontLeft,
        template = Suspension/part/SuspensionFrontRight,
        /* Only the part: a whole-vehicle template line also replaces the vehicle's physics shapes
           and useChassisPhysicsCollision with the template's (VehicleScript.copyPhysicsFrom). */
        template = TrunkDoor/part/TrunkDoor,
        part TireFrontLeft
        {
            table install
            {
                requireInstalled = SuspensionFrontLeft,
            }
        }
        part TireFrontRight
        {
            table install
            {
                requireInstalled = SuspensionFrontRight,
            }
        }
        /* The battery box's lid. Vehicles.ContainerAccess.TruckBed opens a container only while
           the part named TrunkDoor is open or gone, so the lid takes that name, and its lock is
           the box's padlock. It starts closed and often locked (RenewableEnergy.ArrowBoard). No
           key spawns for it, and the lid needs none to come off, so the mechanics window unbolts a
           locked one as the template's trunk lid, with its wrench, time and Mechanics odds; bolt
           cutters can cut the padlock instead (RenewableEnergyAction). A lua
           block replaces the template's whole list. Its model shows only while a lid is
           installed, and the padlock's only while that lid is locked with an uncut lock
           (RenewableEnergy.ArrowBoard). The game swings whichever of a part's models it showed
           first (BaseVehicle.getModelInfoForPart), so the padlock comes after Default, which
           shows it first on install, and only the padlock is ever shown on its own. */
        part TrunkDoor
        {
            area = RenewableEnergyBatteryBox,
            mechanicRequireKey = false,
            model Default
            {
                file = RenewableEnergyArrowBoardTrailerLid,
            }
            model Padlock
            {
                file = RenewableEnergyArrowBoardTrailerPadlock,
            }
            /* The clip runs from open to shut, as vanilla's door clips do. Opened and Closed hold
               it at one end: the game shows them by the door's state whenever no swing is playing
               (BaseVehicle.updateAnimationPlayer). RenewableEnergy.ArrowBoard plays the swings,
               in every game that draws the trailer, when it sees the door change; anim blocks add
               to the template's, so Open and Close keep only their trunk sounds, which the player
               opening or shutting the lid hears. */
            anim Opened
            {
                anim = Lid_closing,
                animate = false,
            }
            anim Closed
            {
                anim = Lid_closing,
                reverse = true,
                animate = false,
            }
            anim SwingOpen
            {
                anim = Lid_closing,
                reverse = true,
                rate = 2.0,
            }
            anim SwingClose
            {
                anim = Lid_closing,
                rate = 2.5,
            }
            /* Getting into a padlocked box, or shutting it again, shouldn't wait on a car
               mechanic's know-how, so the lid comes off and goes back on without the template's
               Basic Mechanics. A table block adds to the template's, and an empty value takes a
               key out (VehicleScript.LoadTable). */
            table install
            {
                recipes = ,
            }
            table uninstall
            {
                recipes = ,
            }
            lua
            {
                create = RenewableEnergy.ArrowBoard.createLid,
                init = RenewableEnergy.ArrowBoard.initLid,
                update = Vehicles.Update.TrunkDoor,
                use = Vehicles.Use.TrunkDoor,
            }
        }
        /* The frame, front and back, under the ids whose condition the game shows on the mask's
           zones (BaseVehicle.doDamageOverlay): the front on Head, the back on Tail and the side
           guards. Crashes and road kills damage them (addDamageFront/Rear, addDamage*HitAChr),
           but only while they hold an item. Hidden from the mechanics window, which would call
           them Hood and Truck Bed; Repair Frame welds them from the trailer's menu instead. With
           no door and no container they open nothing. specificItem = false keeps the item's name
           as written, without the mechanicType that VehicleScript otherwise appends. */
        part EngineDoor
        {
            area = Engine,
            category = nodisplay,
            itemType = RenewableEnergy.TrailerFrame,
            specificItem = false,
            lua
            {
                create = Vehicles.Create.Default,
                init = RenewableEnergy.ArrowBoard.initFrame,
            }
        }
        part TruckBed
        {
            area = TruckBed,
            category = nodisplay,
            itemType = RenewableEnergy.TrailerFrame,
            specificItem = false,
            lua
            {
                create = Vehicles.Create.Default,
                init = RenewableEnergy.ArrowBoard.initFrame,
            }
        }
        /* The lamp board, under the rear windshield's id: the game lights the mask zone of that
           id from the lamps texture while windowLightsOn is set and the part holds an item, with no
           battery part to ask (BaseVehicle.updateLights), so RenewableEnergy.ArrowBoard flashes the
           lamps by that flag alone. The id also shows the part's condition on that zone and takes
           rear crashes (doDamageOverlay, addDamageRear); a board beaten to nothing stays dark until
           Repair Frame welds it, so it holds the frame's item, whose welding that is. */
        part WindshieldRear
        {
            area = TruckBed,
            category = nodisplay,
            itemType = RenewableEnergy.TrailerFrame,
            specificItem = false,
            lua
            {
                create = Vehicles.Create.Default,
                init = RenewableEnergy.ArrowBoard.initBoard,
            }
        }
        part RenewableEnergyPanelLeft
        {
            area = RenewableEnergyPanelLeft,
            mechanicArea = Left,
            model Whole
            {
                file = RenewableEnergyArrowBoardTrailerPanelLeft,
            }
            model LightCracked
            {
                file = RenewableEnergyArrowBoardTrailerPanelLeftLightCracked,
            }
            model Cracked
            {
                file = RenewableEnergyArrowBoardTrailerPanelLeftCracked,
            }
        }
        part RenewableEnergyPanelRight
        {
            area = RenewableEnergyPanelRight,
            mechanicArea = Right,
            model Whole
            {
                file = RenewableEnergyArrowBoardTrailerPanelRight,
            }
            model LightCracked
            {
                file = RenewableEnergyArrowBoardTrailerPanelRightLightCracked,
            }
            model Cracked
            {
                file = RenewableEnergyArrowBoardTrailerPanelRightCracked,
            }
        }
        /* The two panels hold a Solar Module each; the part's condition is the module's. Lua shows
           the Whole, LightCracked or Cracked model by that condition (RenewableEnergy.ArrowBoard), so
           the game mustn't show them all whenever a module is installed. specificItem = false keeps the
           module types as written (see the frame). A pattern only reaches parts already declared,
           so this comes after them. */
        part RenewableEnergyPanel*
        {
            category = bodywork,
            itemType = RenewableEnergy.SolarModule;RenewableEnergy.HomemadeSolarModule,
            specificItem = false,
            setAllModelsVisible = false,
            table install
            {
                items
                {
                    1
                    {
                        tags = base:wrench,
                        count = 1,
                        keep = true,
                        equip = primary,
                    }
                }
                time = 200,
                skills = Mechanics:2;Electricity:1,
                test = Vehicles.InstallTest.Default,
                complete = RenewableEnergy.ArrowBoard.panelChanged,
            }
            table uninstall
            {
                items
                {
                    1
                    {
                        tags = base:wrench,
                        count = 1,
                        keep = true,
                        equip = primary,
                    }
                }
                time = 200,
                skills = Mechanics:2;Electricity:1,
                test = Vehicles.UninstallTest.Default,
                complete = RenewableEnergy.ArrowBoard.panelChanged,
            }
            lua
            {
                create = RenewableEnergy.ArrowBoard.createPanel,
                init = RenewableEnergy.ArrowBoard.initPanel,
            }
        }
        part RenewableEnergyBatteryBox
        {
            area = RenewableEnergyBatteryBox,
            category = bodywork,
            mechanicArea = Back,
            container
            {
                capacity = 100,
                test = Vehicles.ContainerAccess.TruckBed,
            }
        }
        attachment trailer
        {
            offset = 0.0 -0.16 1.485,
            rotate = 0.0 0.0 0.0,
            canAttach = trailer,
            updateconstraint = false,
        }
    }

    /* Repair Frame: welding, as vanilla's Fix Trailer Welding welds its trailers' bodies. */
    fixing Fix Trailer Frame Welding
    {
        Require = RenewableEnergy.TrailerFrame,
        GlobalItem = Base.BlowTorch=2,
        ConditionModifier = 1.2,
        Fixer = Base.SheetMetal; MetalWelding=3;Mechanics=2,
        Fixer = Base.SmallSheetMetal=2; MetalWelding=3;Mechanics=2,
    }
}
