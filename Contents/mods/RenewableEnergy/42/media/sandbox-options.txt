/*
    Sunlight multiplier: 1 is what 1993 modules really made in Kentucky. The default puts a
    fridge-freezer at four or five modules in an average July and about eight on racks in
    December, so that each one you find matters.
*/

	page = RenewableEnergy, translation = RenewableEnergy_SunlightMultiplier,
	type = double, min = 1, max = 5, default = 2.5,
}
