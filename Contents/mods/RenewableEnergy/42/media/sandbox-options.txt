VERSION = 1,
/*
    Sunlight multiplier: 1 is Kentucky's real sunlight, as many modules to a fridge as in 1993.
    The default puts a fridge-freezer at four or five modules in an average July and about eight
    on racks in December, so that each one you find matters.
*/
option RenewableEnergy.SunlightMultiplier {
	page = RenewableEnergy, translation = RenewableEnergy_SunlightMultiplier,
	type = double, min = 1, max = 5, default = 2.5,
}
/*
    Biogas multiplier: 1 is what a real digester makes from the feed. The game drops far less
    manure than real animals do, so the default makes a kitchen-scraps barrel worth keeping.
*/
option RenewableEnergy.BiogasMultiplier {
	page = RenewableEnergy, translation = RenewableEnergy_BiogasMultiplier,
	type = double, min = 1, max = 10, default = 3,
}
option RenewableEnergy.ChargeFreq {
	page = RenewableEnergy, translation = RenewableEnergy_ChargeFreq, valueTranslation = RenewableEnergy_ChargeFreq,
	type = enum, numValues = 2, default = 1,
}
/* Percent of the wear batteries take from deep discharges. */
option RenewableEnergy.BatteryWear {
	page = RenewableEnergy, translation = RenewableEnergy_BatteryWear,
	type = integer, min = 0, max = 1000, default = 100,
}
/* Bank charge, in percent, below which a failsafe starts its generator and at which it stops it. */

option RenewableEnergy.FailsafeStopCharge {

/* Percent chance that each named solar site is placed, rolled once per world; the few marked
